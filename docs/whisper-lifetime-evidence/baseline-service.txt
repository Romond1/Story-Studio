import { Worker } from 'node:worker_threads';
import { spawn, type ChildProcess } from 'node:child_process';
import { promises as fs } from 'node:fs';
import path from 'node:path';
import { hasSpeech, type JapaneseMode, type ReadingToken, type TranslationRequest, type TranslationResult, type SpeechBackend } from '../../shared/translation';
function getWhisperBackend() {
  return require('./whisper.cjs') as {probeNvidia:()=>Promise<{device?:string;reason?:string}>;verifyCudaLog:(log:string)=>boolean;cudaDeviceFromLog:(log:string)=>string|undefined};
}

export class TranslationService {
  private worker: Worker | null = null;
  private disposal: Promise<void> | null = null;
  private child: ChildProcess | null = null;
  private nativeClosed: Promise<void> | null = null;
  private pending: {resolve:(v:any)=>void;reject:(e:Error)=>void;timer:NodeJS.Timeout;recoverable:boolean} | null = null;
  private speechReject: ((e:Error)=>void) | null = null;
  private revision = 0;
  private serial = 0;
  private target = '';
  private busy = false;
  constructor(private cache: string, private status: (s:string)=>void, private workerFile = path.join(__dirname,'inference.cjs')) {}
  cancelCurrent(unload=false){if(unload||this.busy||this.pending||this.child)this.cancel();}
  cancel() {
    this.revision++;
    this.pending?.reject(new Error('Cancelled')); if(this.pending)clearTimeout(this.pending.timer); this.pending=null;
    this.speechReject?.(new Error('Cancelled')); this.speechReject=null;
    const child=this.child, nativeClosed=this.nativeClosed;this.child=null;this.nativeClosed=null;
    const worker=this.worker; this.worker=null;
    if(worker||nativeClosed){
      const disposal=Promise.all([this.disposal,nativeClosed,worker?.terminate().catch(()=>{})]).then(()=>{});
      this.disposal=disposal;
      void disposal.then(()=>{if(this.disposal===disposal)this.disposal=null;});
    }
    // Hold new model preparation until the old Windows executable releases its files.
    child?.kill();
    this.target=''; this.busy=false;
  }
  private failWorker(error:Error) {
    const pending=this.pending;
    if(!pending?.recoverable){pending?.reject(error);this.cancel();return;}
    // A failed speech preparer is replaceable. Preserve the active request's
    // revision so Auto can retry CPU; only an explicit cancel invalidates it.
    this.pending=null;clearTimeout(pending.timer);
    const worker=this.worker;this.worker=null;
    if(worker){
      const disposal=Promise.all([this.disposal,worker.terminate().catch(()=>{})]).then(()=>{});
      this.disposal=disposal;
      void disposal.then(()=>{if(this.disposal===disposal)this.disposal=null;});
    }
    pending.reject(error);
  }
  private async call(payload: Record<string,unknown>): Promise<any> {
    const revision=this.revision;
    if(this.disposal)await this.disposal;
    if(revision!==this.revision)throw new Error('Cancelled');
    if(this.pending)return Promise.reject(new Error('Translation is busy'));
    if(!this.worker){
      const worker=new Worker(this.workerFile,{workerData:{cache:this.cache}});this.worker=worker;
      worker.on('message',message=>{
        if(worker!==this.worker)return;
        if(message.status){this.status(message.status);return;}
        if(message.id!==this.serial||!this.pending)return;
        const pending=this.pending; this.pending=null; clearTimeout(pending.timer);
        if(message.error){pending.reject(new Error(message.error));if(!pending.recoverable)this.cancel();}else pending.resolve(message);
      });
      const fail=(error:Error)=>{if(worker!==this.worker)return;this.failWorker(error);};
      worker.on('error',fail);worker.on('exit',code=>fail(new Error(`Local inference worker exited (${code})`)));
    }
    return new Promise((resolve,reject)=>{
      const timer=setTimeout(()=>this.failWorker(new Error('Local model operation timed out; retry when ready.')),360000);
      this.pending={resolve,reject,timer,recoverable:payload.kind==='speech'};this.worker!.postMessage({...payload,id:++this.serial});
    });
  }
  async run(request: TranslationRequest): Promise<TranslationResult> {
    if(this.busy)throw new Error('Finish or cancel the current utterance first.');
    if(!request||!['ja','it','ro'].includes(request.target)||!['base.en','small.en'].includes(request.model))throw new Error('Invalid translation settings');
    if(request.execution!==undefined&&!['auto','gpu','cpu'].includes(request.execution))throw new Error('Invalid speech execution setting');
    if(request.audio!==undefined && (!Array.isArray(request.audio)||!hasSpeech(request.audio)))throw new Error('No clear speech detected. Speak closer to the microphone and try again.');
    if(request.audio===undefined && (typeof request.text!=='string'||!request.text.trim()||request.text.length>2000))throw new Error('Enter 1–2000 English characters.');
    if(this.target && this.target!==request.target)this.cancel();
    this.target=request.target;this.busy=true;
    const revision=this.revision,started=performance.now(),metrics:Record<string,number>={};
    const check=()=>{if(revision!==this.revision)throw new Error('Cancelled');};
    try{
      let transcript=request.text?.trim() ?? '';
      let speechBackend:SpeechBackend|undefined;
      if(request.audio){
        const execution=request.execution??'auto';
        let device:string|undefined,reason:string|undefined;
        if(execution!=='cpu'){
          const probe=await getWhisperBackend().probeNvidia();check();device=probe.device;reason=probe.reason;
          if(!device&&execution==='gpu')throw new Error(`${reason} Select CPU or Auto, or install a compatible NVIDIA driver.`);
        }
        const attempt=async(backend:'cpu'|'cuda')=>{
          check();const prepStart=performance.now();
          const prepared=await this.call({kind:'speech',model:request.model,backend});check();
          metrics.speechInstallVerifyMs=(metrics.speechInstallVerifyMs??0)+performance.now()-prepStart;
          return this.transcribe(request.audio!,{...prepared.result,backend});
        };
        let speech;
        if(device){
          try{speech=await attempt('cuda');check();speechBackend={backend:'cuda',...(speech.device?{device:speech.device}:{})};}
          catch(error){
            check();reason=error instanceof Error?error.message:String(error);
            if(execution==='gpu')throw new Error(`GPU transcription failed: ${reason} Select CPU or Auto, or update your NVIDIA driver.`);
            this.status(`GPU unavailable; using CPU. ${reason}`);
          }
        }
        if(!speech){speech=await attempt('cpu');check();speechBackend={backend:'cpu',...(reason?{reason}:{})};}
        transcript=speech.text;Object.assign(metrics,speech.metrics);
      }
      if(!transcript.trim())throw new Error('No speech recognized. Try again.');
      const translated=await this.call({kind:'translate',text:transcript,target:request.target});check();
      Object.assign(metrics,translated.metrics,{translationMs:translated.result.translationMs,translationLoadMs:translated.result.loadMs,totalMs:performance.now()-started});
      return {transcript,translation:translated.result.text,metrics,...(speechBackend?{speechBackend}:{})};
    }finally{if(revision===this.revision)this.busy=false;}
  }
  async format(text:string,mode:JapaneseMode):Promise<ReadingToken[]>{
    if(typeof text!=='string'||text.length>2000||!['standard','kana','furigana'].includes(mode))throw new Error('Invalid Japanese text');
    if(this.busy)throw new Error('Wait for translation to finish.');
    this.busy=true;const revision=this.revision;
    try{return (await this.call({kind:'format',text,mode})).result;}finally{if(revision===this.revision)this.busy=false;}
  }
  private async transcribe(audio:number[],prepared:{exe:string;modelPath:string;backend:'cpu'|'cuda'}){
    const revision=this.revision;
    const preprocessStarted=performance.now();
    this.status(`Transcribing locally (${prepared.backend==='cuda'?'NVIDIA CUDA':'CPU'})…`);
    await fs.mkdir(this.cache,{recursive:true});
    const folder=await fs.mkdtemp(path.join(this.cache,'utterance-'));
    const file=path.join(folder,'input.wav'),out=path.join(folder,'result');
    const wav=Buffer.alloc(44+audio.length*2);
    wav.write('RIFF');wav.writeUInt32LE(wav.length-8,4);wav.write('WAVEfmt ',8);wav.writeUInt32LE(16,16);wav.writeUInt16LE(1,20);wav.writeUInt16LE(1,22);wav.writeUInt32LE(16000,24);wav.writeUInt32LE(32000,28);wav.writeUInt16LE(2,32);wav.writeUInt16LE(16,34);wav.write('data',36);wav.writeUInt32LE(audio.length*2,40);
    audio.forEach((v,i)=>wav.writeInt16LE(Math.round(Math.max(-1,Math.min(1,v))*32767),44+i*2));
    let childClosed:Promise<void>|undefined;
    try{
      if(revision!==this.revision)throw new Error('Cancelled');
      await fs.writeFile(file,wav);
      if(revision!==this.revision)throw new Error('Cancelled');
      const started=performance.now();
      const stderr=await new Promise<string>((resolve,reject)=>{
        const child=spawn(prepared.exe,['-m',prepared.modelPath,'-f',file,'-l','en','-t','2',...(prepared.backend==='cpu'?['-ng']:[]),'-nt','-oj','-of',out],{windowsHide:true,cwd:path.dirname(prepared.exe),stdio:['ignore','ignore','pipe']});
        childClosed=new Promise(resolve=>child.once('close',()=>resolve()));
        this.nativeClosed=childClosed;
        this.child=child;this.speechReject=reject;let log='';
        const timer=setTimeout(()=>{child.kill();reject(new Error('Transcription timed out.'));},120000);
        child.stderr?.on('data',data=>{log=(log+data.toString()).slice(-24000);});
        child.on('error',error=>{clearTimeout(timer);reject(error);});
        child.on('close',code=>{clearTimeout(timer);if(this.child===child){this.child=null;this.nativeClosed=null;this.speechReject=null;}code===0?resolve(log):reject(new Error(`Whisper failed (${code}). ${log.slice(-500)}`));});
      });
      if(revision!==this.revision)throw new Error('Cancelled');
      if(prepared.backend==='cuda')getWhisperBackend().verifyCudaLog(stderr);
      const json=JSON.parse(await fs.readFile(out+'.json','utf8'));
      const text=(json.transcription ?? []).filter((s:any)=>s.no_speech_prob===undefined||s.no_speech_prob<0.6).map((s:any)=>s.text).join(' ').replace(/\[(?:BLANK_AUDIO|SILENCE|MUSIC)\]/gi,'').trim();
      const timing=(name:string)=>Number(stderr.match(new RegExp(`whisper_print_timings:\\s*${name} time\\s*=\\s*([\\d.]+)`))?.[1]??0);
      return {text,device:prepared.backend==='cuda'?getWhisperBackend().cudaDeviceFromLog(stderr):undefined,metrics:{speechPreprocessMs:started-preprocessStarted,transcriptionMs:performance.now()-started,whisperLoadMs:timing('load'),whisperNativeTotalMs:timing('total'),whisperEncodeMs:timing('encode'),whisperDecodeMs:timing('decode'),whisperBatchDecodeMs:timing('batchd')}};
    }finally{
      // Windows may keep input/output files locked until the native process exits.
      // Cancellation rejects immediately, but private utterance cleanup must wait.
      await childClosed;
      await fs.rm(folder,{recursive:true,force:true}).catch(()=>{});
    }
  }
}
